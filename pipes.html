function applyPathPipes(path) { for (let i = 1; i < path.length - 1; i++) { let prev = path[i-1]; let curr = path[i]; let next = path[i+1]; let d1 = getDir(curr, prev); let d2 = getDir(curr, next); let cell = grid[curr.r][curr.c]; // Deduce ideal pipe shape if ((d1 % 2) === (d2 % 2)) { cell.type = 'I'; cell.rotation = d1 % 2 === 0 ? 0 : 1; } else { cell.type = 'L'; if ((d1===0 && d2===1) || (d1===1 && d2===0)) cell.rotation = 0; if ((d1===1 && d2===2) || (d1===2 && d2===1)) cell.rotation = 1; if ((d1===2 && d2===3) || (d1===3 && d2===2)) cell.rotation = 2; if ((d1===3 && d2===0) || (d1===0 && d2===3)) cell.rotation = 3; } } // Orientar correctamente las tuberías estáticas de inicio y fin if (path.length > 1) { // Tubería de inicio let startCell = grid[path[0].r][path[0].c]; let startDir = getDir(path[0], path[1]); // Dirección hacia el primer paso startCell.rotation = (startDir === 0 || startDir === 2) ? 0 : 1; // Tubería de fin let endCell = grid[path[path.length - 1].r][path[path.length - 1].c]; let endDir = getDir(path[path.length - 1], path[path.length - 2]); // Dirección hacia el último paso endCell.rotation = (endDir === 0 || endDir === 2) ? 0 : 1; } }
